# 特征映射
feature_desc = {
    '1': '有毛发', '2': '产奶', '3': '有羽毛', '4': '不会飞', '5': '会下蛋',
    '6': '吃肉', '7': '有犬齿', '8': '有爪', '9': '眼盯前方', '10': '有蹄',
    '11': '反刍', '12': '黄褐色', '13': '有斑点', '14': '有黑色条纹', '15': '长脖',
    '16': '长腿', '17': '不会飞', '18': '会游泳', '19': '黑白二色', '20': '善飞',
    '21': '哺乳类', '22': '鸟类', '23': '食肉类', '24': '蹄类'
}

# 规则库： (前提集合, 结论)
rules = [
    ({'1'}, '21'),   # 有毛发 -> 哺乳类
    ({'2'}, '21'),   # 产奶 -> 哺乳类
    ({'3'}, '22'),   # 有羽毛 -> 鸟类
    ({'4', '5'}, '22'),  # 不会飞 & 会下蛋 -> 鸟类
    ({'6', '21'}, '23'), # 吃肉 & 哺乳类 -> 食肉类
    ({'7', '8', '9'}, '23'), # 有犬齿 & 有爪 & 眼盯前方 -> 食肉类
    ({'10', '21'}, '24'), # 有蹄 & 哺乳类 -> 蹄类
    ({'11', '21'}, '24'), # 反刍 & 哺乳类 -> 蹄类
]

# 动物库：动物名 -> 所需特征集合
animals = {
    '金钱豹': {'12', '13', '21', '23'},
    '虎': {'12', '14', '21', '23'},
    '斑马': {'14', '24'},
    '长颈鹿': {'13', '15', '16', '24'},  # 原规则没有14，但描述有黑色条纹，此处按原规则
    '鸵鸟': {'4', '15', '16', '22'},
    '企鹅': {'4', '18', '19', '22'},
    '信天翁': {'20', '22'}
}

def inference(initial_facts):
    facts = set(initial_facts)
    changed = True
    while changed:
        changed = False
        for premises, conclusion in rules:
            if premises.issubset(facts) and conclusion not in facts:
                facts.add(conclusion)
                print(f"推理：{', '.join(feature_desc[p] for p in premises)} -> {feature_desc[conclusion]}")
                changed = True
    return facts

def match_animal(facts):
    candidates = []
    for animal, required in animals.items():
        if required.issubset(facts):
            candidates.append(animal)
    return candidates

def main():
    print('''输入对应条件前面的数字:
            *******************************************************
            *1:有毛发  2:产奶  3:有羽毛  4:不会飞  5:会下蛋          *
            *6:吃肉  7:有犬齿  8:有爪  9:眼盯前方  10:有蹄         *
            *11:反刍  12:黄褐色  13:有斑点  14:有黑色条纹  15:长脖 *
            *16:长腿  17:不会飞  18:会游泳  19:黑白二色  20:善飞   *
            *21：哺乳类  22:鸟类  23:食肉类  24：蹄类              *
            *******************************************************
            *******************当输入数字0时!程序结束***************
    ''')
    user_inputs = []
    while True:
        num = input("请输入：").strip()
        if num == '0':
            break
        if num in feature_desc:
            user_inputs.append(num)
        else:
            print("无效输入，请重新输入。")
    if not user_inputs:
        print("未输入任何特征。")
        return
    print("\n前提条件为：")
    print(' '.join(feature_desc[num] for num in user_inputs))
    print("\n推理过程如下：")
    facts = inference(user_inputs)
    print("\n最终事实：", ', '.join(feature_desc[f] for f in sorted(facts, key=int)))
    animals_found = match_animal(facts)
    if len(animals_found) == 1:
        print(f"\n所识别的动物为{animals_found[0]}")
    elif len(animals_found) > 1:
        print(f"\n根据所给条件匹配到多个动物：{', '.join(animals_found)}，请提供更多特征。")
    else:
        print("\n根据所给条件无法判断为何种动物")

if __name__ == '__main__':
    main()
